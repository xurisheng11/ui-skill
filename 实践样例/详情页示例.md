# 详情页示例

包含页面标题、基本信息键值对展示的详情页布局。

```json
{
  "status": 0,
  "message": "操作成功",
  "data": {
    "aws": {
      "#type": "aws_wrapper",
      "#children": [
        {
          "#type": "aws_space_between",
          "#direction": "vertical",
          "#size": "m",
          "#content": {
            "apigroup_title": {
              "#type": "aws_header",
              "#title": "详情页基本信息",
              "#variant": "h1"
            },
            "aws_container": {
              "#type": "aws_container",
              "#header": {
                "#type": "aws_header",
                "#title": "基本信息",
                "#actions": {
                  "edit": {
                    "#type": "aws_button",
                    "#text": "编辑",
                    "#href": "#"
                  }
                }
              },
              "#content": {
                "#type": "aws_key_value_pairs",
                "#columns": 4,
                "#items": [
                  {"label": "API 组描述", "value": "主服务"},
                  {"label": "描述", "value": "测试平台的主服务"},
                  {
                    "label": "API 源",
                    "value": {
                      "#type": "aws_box",
                      "#content": {
                        "info": {"#markup": "[taas-api] Taas后端主服务"},
                        "document": {
                          "#type": "aws_link",
                          "#content": "查看文档",
                          "#href": "#",
                          "#external": true
                        }
                      }
                    }
                  },
                  {
                    "label": "创建时间",
                    "value": {
                      "#type": "aws_box",
                      "#content": {
                        "#theme": "time_format",
                        "time": "2023-07-04T03:11:25.249Z"
                      }
                    }
                  },
                  {"label": "上次同步时间", "value": "2022-11-10 15:30:00"}
                ]
              }
            }
          }
        }
      ]
    }
  }
}
```
