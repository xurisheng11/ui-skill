# 条件显示表单示例

利用 aws_box 的 display 属性实现表单内容的条件显示/隐藏。

> **注意**：此显示隐藏仅为前端页面的显示隐藏，若隐藏的内容中存在表单校验，依然会被校验，且表单提交中依然包括此部分内容的数据。

## 方法1：使用 display 属性隐藏

```json
{
  "status": 0,
  "message": null,
  "data": {
    "aws": {
      "#type": "aws_wrapper",
      "#children": [
        {
          "topic_type": {
            "#type": "aws_form_field",
            "#label": "话题类型",
            "#control": {
              "topic_type": {
                "#type": "aws_radio_group",
                "#items": [
                  {"label": "持久化", "value": "PERSISTENT"},
                  {"label": "非持久化", "value": "NONPERSISTENT"}
                ],
                "#default_value": "PERSISTENT"
              }
            }
          },
          "message_expire_time": {
            "#type": "aws_box",
            "#display": {"#markup": "value_topic_type=='PERSISTENT'?'none':'block'"},
            "#content": {
              "#type": "aws_form_field",
              "#label": "消息过期时间",
              "#constraint_text": "应介于60秒 至 15 天之间",
              "#description": "未消费消息的过期时间，超时未 ACK 则跳过对该消息的处理",
              "#control": {
                "message_expire_time": {
                  "#type": "aws_space_between",
                  "#direction": "horizontal",
                  "#size": "xs",
                  "content": {
                    "time": {
                      "#type": "aws_input",
                      "#input_type": "number",
                      "#default_value": "12"
                    },
                    "unit": {
                      "#type": "aws_select",
                      "#select_options": [
                        {"label": "秒", "value": "SECOND"},
                        {"label": "分", "value": "MINUTE"},
                        {"label": "时", "value": "HOUR"},
                        {"label": "天", "value": "DAY"}
                      ],
                      "#default_value": {"label": "时", "value": "HOUR"}
                    }
                  }
                }
              }
            }
          }
        }
      ]
    }
  }
}
```

## 方法2：使用三元组控制显示

与方法1的区别：隐藏时不存在前端结构，在存在父元素 space_between 时隐藏后不产生间距。

```json
{
  "status": 0,
  "message": null,
  "data": {
    "aws": {
      "#type": "aws_wrapper",
      "#children": {
        "#type": "aws_space_between",
        "#direction": "vertical",
        "#size": "s",
        "#content": [
          {
            "#type": "aws_form_field",
            "#label": "话题类型",
            "#control": {
              "topic_type": {
                "#type": "aws_radio_group",
                "#items": [
                  {"label": "持久化", "value": "PERSISTENT"},
                  {"label": "非持久化", "value": "NONPERSISTENT"}
                ],
                "#default_value": "PERSISTENT"
              }
            }
          },
          {"#markup": "{value_topic_type=='PERSISTENT'?'':"},
          {
            "#type": "aws_form_field",
            "#label": "消息过期时间",
            "#control": {
              "message_expire_time": {
                "#type": "aws_space_between",
                "#direction": "horizontal",
                "#size": "xs",
                "#content": {
                  "time": {
                    "#type": "aws_input",
                    "#input_type": "number",
                    "#default_value": "12"
                  },
                  "unit": {
                    "#type": "aws_select",
                    "#select_options": [
                      {"label": "秒", "value": "SECOND"},
                      {"label": "分", "value": "MINUTE"},
                      {"label": "时", "value": "HOUR"},
                      {"label": "天", "value": "DAY"}
                    ],
                    "#default_value": {"label": "时", "value": "HOUR"}
                  }
                }
              }
            }
          },
          {"#markup": "}"},
          {
            "#type": "aws_form_field",
            "#label": "input",
            "#control": {"input": {"#type": "aws_input"}}
          }
        ]
      }
    }
  }
}
```
