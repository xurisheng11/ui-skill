# 表单验证示例

带有必填校验和正则校验的表单，支持多种表单控件类型的验证。

  "message": "操作成功",
  "data": {
    "aws": {
      "#type": "aws_wrapper",
      "#children": [
        {
          "#type": "aws_form_field",

          "#label": "label",
          "#info": "info",
          "#required": true,
          "#require_err": "请填写此字段",
          "#pattern": "^[^\\s@]+@[^\\s@]+\\\\.[^\\s@]+$",
          "#pattern_err": "pattern_err",
          "#constraint_text": "constraint_text",
          "#control": {
            "input2": {"#type": "aws_input"}
          }
        },
        {
          "#type": "aws_form_field",
          "#description": "description",
          "#label": "label",
          "#info": "info",
          "#required": true,

          "#control": {
            "input6": {
              "#type": "aws_radio_group",
              "#items": [
                {"value": "first", "label": "First choice"},
                {"value": "second", "label": "Second choice"},
