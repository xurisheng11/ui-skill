# 实践样例：表单页完整模板

包含创建/编辑两种模式的完整表单页（带取消和提交按钮）。

## TypeScript 实现（Controller）

```typescript
@Controller('/module', { tagName: 'module', description: '创建/编辑模块' })
export class ModuleFormController extends BaseController {
  private name = 'module';

  @Get('/form/build')
  public async build(@Query('id') id: string) {
    const isEdit = !!id;
    const linkPrefix = id ? `/module/detail/${id}` : '/module';
    
    let data: any = {};
    if (isEdit) {
      data = await this.get(`${Base.ROOT}/modules/${id}`);
    }
    
    // 右侧按钮区
    const buttonContent = {
      button_cancel: BaseUI.linkButtonLight('取消', this.getUrl(linkPrefix)),
      button_save: BaseUI.submitButton(isEdit ? '保存' : '创建'),
    };
    
    // 表单字段
    const itemContent = {
      // 仅创建时显示名称（编辑时不可改）
      item_name: !isEdit && BaseUI.textField(
        'name', '名称', '请输入名称',
        BaseUI.validatePattern.identifier.message,
        data.name,
        { pattern: BaseUI.validatePattern.identifier.pattern },
      ),
      
      // 描述（可选）
      item_description: BaseUI.textArea(
        'description', '描述 - 可选', '',
        BaseUI.validatePattern.description.message,
        data.description,
        { pattern: BaseUI.validatePattern.description.pattern },
        false, // required=false
      ),
      
      // 下拉选择
      item_type: BaseUI.selectField(
        'type', '类型',
        [
          { label: '类型A', value: 'TYPE_A' },
          { label: '类型B', value: 'TYPE_B' },
        ],
        data.type ? { label: data.type, value: data.type } : undefined,
      ),
    };
    
    return Base.successWrapper([
      BaseUI.spaceBetween(
        {
          header: BaseUI.pageTitle('title', isEdit ? '编辑模块' : '创建模块'),
          info: BaseUI.container(BaseUI.spaceBetween(itemContent), '基本信息'),
        },
        buttonContent, // 右侧按钮
      ),
    ]);
  }

  @Post('/form/submit')
  public async submit(@Body() body: any) {
    let id = body.id;
    
    const pData = {
      name: body.name,
      description: body.description,
      type: this.selectValue(body.type), // 解析 select 值
    };
    
    if (id) {
      await this.put(`${Base.ROOT}/modules/${id}`, { data: pData });
    } else {
      const result = await this.post(`${Base.ROOT}/modules`, { data: pData });
      id = result.id;
    }
    
    return Base.success(this.getUrl(`/module/detail/${id}`));
  }

  @Get('/form')
  public async getFormUris() { return Base.successForm(); }
  
  @Get('/form/formId')
  public async formId() { return Base.success(this.name + '_form'); }
  
  @Post('/form/validate')
  public async validate(@Body() body: any) {
    // 校验名称唯一性（仅创建时）
    if (body.name) {
      const existing = await this.get(`${Base.ROOT}/modules?name=${body.name}`);
      if (existing && existing.length > 0) {
        return Base.success({ errors: { name: '名称已存在，请换一个' } });
      }
    }
    return Base.success(null);
  }
}
```
