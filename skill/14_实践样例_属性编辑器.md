# 实践样例：属性编辑器（动态键值对列表）

使用 `aws_attribute_editor` 构建可动态添加/删除行的键值对列表。

## TypeScript 实现（使用 BaseUI.attributeEditorAws）

```typescript
// 基础键值对（名称 + 值）
BaseUI.attributeEditorAws(
  [
    {
      '#label': '名称',
      '#key': 'name',
      '#control': { '#type': 'aws_input' },
    },
    {
      '#label': '值',
      '#key': 'value',
      '#control': { '#type': 'aws_input' },
    },
  ],
  data.envs || [],      // 默认数据
  '',                   // 空状态提示
  false,                // disable_add_button
  false,                // hide_add_button（只读时设为 true）
)
```

## 只读模式（禁用添加/删除）

```typescript
BaseUI.attributeEditorAws(
  fields,
  data.envs || [],
  '',
  true,   // disable_add_button = true
  true,   // hide_add_button = true（完全隐藏添加按钮）
)
```

## 混合控件类型（多种输入控件）

      '#label': '类型',

      '#control': BaseUI.selectAws([
        { label: '字符串', value: 'string' },
        { label: '整数',   value: 'int' },
        { label: '密码',   value: 'password' },
      ]),
    },
    {
      '#label': '是否必填',
      '#key': 'required',
      '#control': BaseUI.selectAws([
        { label: '是', value: '1' },
        { label: '否', value: '0' },
      ]),
    },
    {
      '#label': '启用',
      '#key': 'enabled',
      '#control': { '#type': 'aws_toggle', '#label': '启用' },
    },
  ],
  defaultData,
)
```

## 使用 grid_layout 自定义列宽

  '#type': 'aws_attribute_editor',
  '#default_value': [{ key: '', value: '', label: '', type: '' }],
  '#add_button_text': '添加',

  ],

```

## 在 submit 中解析数据

```typescript
// 属性编辑器提交数据为 JSON 字符串
const envVarsStr = body.envs;
let envVars: any[] = [];
try {
  envVars = JSON.parse(envVarsStr || '[]');
  // 过滤掉空行（空数组套空数组的情况）
  if (envVars.length === 1 && Array.isArray(envVars[0]) && envVars[0].length === 0) {
    envVars = [];
  }
} catch(e) {
  // 解析失败处理
}

// 过滤空行
envVars = envVars.filter(item => item.name || item.value);
```

## 在 validate 中校验

```typescript
public async validate(@Body() body: any) {
  const envVarsStr = body.envs;
  if (envVarsStr) {
    let parsed: any[];
    try { parsed = JSON.parse(envVarsStr); } catch(e) {
      return Base.success({ errors: { error: '格式不正确' } });
    }
    
    // 过滤有效数据（忽略完全空行）
    const valid = parsed.filter(item => !Array.isArray(item));
    
    for (const item of valid) {
      if (!item.name?.trim()) {
        return Base.success({ errors: { error: '名称不能为空' } });
      }
      if (item.name.length > 255) {
        return Base.success({ errors: { error: `名称【${item.name}】不能超过255个字符` } });
      }
    }
  }
  return Base.success(null);
}
```
