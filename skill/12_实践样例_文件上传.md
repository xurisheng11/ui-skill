# 实践样例：文件上传

通过 useEffect + input[type=file] + ajax-callback 实现文件上传（以 base64 方式传输）。

## 实现要点

1. `const_define` 定义触发器状态（trigger 递增触发上传）
2. `useEffect` 监听 trigger 变化，弹出文件选择框
3. 读取文件内容为 base64，通过 ajax-callback 的 POST 上传
4. 后端接收 base64，解码后存储到 S3

## 前端 JS 代码（完整模板）

```javascript
// 监听触发器变化，弹出文件选择框
useEffect(() => {
  if (!upload_trigger) return;  // trigger 初始值为 0，不触发
  
  // 创建或复用隐藏 input
  var input = document.getElementById('my_upload_input');
  if (!input) {
    input = document.createElement('input');
    input.id = 'my_upload_input';
    input.type = 'file';
    input.accept = '.zip';          // 限制文件类型
    input.style.display = 'none';
    document.body.appendChild(input);
  }
  
  // 每次重新绑定 onchange，避免闭包旧值
  input.onchange = function(e) {
    var file = e.target.files && e.target.files[0];
    if (!file) return;
    
    // 校验文件类型
    if (!file.name.toLowerCase().endsWith('.zip')) {
      setValue_upload_error('仅支持 .zip 文件');
      setValue_upload_status('error');
      input.value = '';
      return;
    }
    
    // 校验文件大小（不超过 10MB）
    if (file.size > 10 * 1024 * 1024) {
      setValue_upload_error('文件大小不超过 10M');
      setValue_upload_status('error');
      input.value = '';
      return;
    }
    
    // 读取文件为 base64
    setValue_upload_status('uploading');
    setValue_upload_error('');
    
    var reader = new FileReader();
    reader.onload = function(ev) {
      var dataUrl = ev.target.result;
      var base64 = dataUrl.indexOf(',') >= 0 ? dataUrl.split(',')[1] : dataUrl;
      
      var bodyStr = JSON.stringify({
        id: 'TARGET_ID',
        dir: 'TARGET_DIR/',
        fileBase64: base64
      });
      
      // POST 到 /ajax-callback
      var params = 'method=post'
        + '&url=' + encodeURIComponent('{LUBAN_HOST}/module/upload')
        + '&body=' + encodeURIComponent(bodyStr)
        + '&content_type=' + encodeURIComponent('application/json');
      
      var xhr = new XMLHttpRequest();
      xhr.open('POST', '/ajax-callback', true);
      xhr.setRequestHeader('Content-Type', 'application/x-www-form-urlencoded');
      xhr.onload = function() {
        var resp;
        try { resp = JSON.parse(xhr.responseText); } catch(ignored) {}
        if (xhr.status >= 200 && xhr.status < 300 && resp && resp.status === 0) {
          setValue_upload_status('success');
          setValue_upload_error('');
        } else {
          setValue_upload_status('error');
          setValue_upload_error((resp && resp.message) || '上传失败，请重试');
        }
        input.value = '';
      };
      xhr.onerror = function() {
        setValue_upload_status('error');
        setValue_upload_error('上传失败，请重试');
        input.value = '';
      };
      xhr.send(params);
    };
    reader.onerror = function() {
      setValue_upload_status('error');
      setValue_upload_error('文件读取失败，请重试');
      input.value = '';
    };
    reader.readAsDataURL(file);
  };
  input.click();
}, [upload_trigger]);

// 监听状态变化，更新提示 DOM
useEffect(() => {
  var statusEl = document.getElementById('upload_status_span');
  if (statusEl) {
    if (upload_status === 'uploading') {
      statusEl.textContent = '上传中...';
      statusEl.style.color = '#888';
    } else if (upload_status === 'error') {
      statusEl.textContent = upload_error || '上传失败';
      statusEl.style.color = '#d13212';
    } else if (upload_status === 'success') {
      statusEl.textContent = '上传成功';
      statusEl.style.color = '#1d8348';
    } else {
      statusEl.textContent = '';
    }
  }
}, [upload_status, upload_error]);
```

## const_define 状态定义

```typescript
[
  { value: 'upload_trigger', set_value: 'setValue_upload_trigger', default: 0 },
  { value: 'upload_status',  set_value: 'setValue_upload_status',  default: '' },
  { value: 'upload_error',   set_value: 'setValue_upload_error',   default: '' },
]
```

## 后端接收接口（Controller）

```typescript
@Post('/upload')
public async upload(@Body() body: { id: string; dir: string; fileBase64: string }) {
  // 解析嵌套 body（ajax-callback 传输时可能有包装）
  let actualBody = body;
  if (!body.id && typeof (body as any).body === 'string') {
    try { actualBody = JSON.parse((body as any).body); } catch(e) {}
  }
  
  // 解码 base64 → Buffer → 上传 S3
  const fileBuffer = Buffer.from(actualBody.fileBase64, 'base64');
  const ossKey = actualBody.dir + 'code.zip';
  await this.putObjectToS3(ossKey, fileBuffer);
  
  // 通知业务接口更新
  await this.post(`${Base.ROOT}/module/${actualBody.id}/code`, {
    data: { ossKey },
  });
  
  return Base.success(null);
}
```
