# 实践样例：列表页完整模板

包含表格、刷新/创建/删除按钮、搜索过滤的完整列表页。

## TypeScript 实现（Controller）

```typescript
@Controller('/module', { tagName: 'module', description: '模块列表' })
export class ModuleController extends BaseController {
  private name = 'module_list';

  @Get('/list/build')
  public async build() {
    const linkPrefix = '/module';
    
    // 列头定义
    const header = [
      BaseUI.tableHeader(
        'name', '名称',
        BaseUI.tableLink('{item.name}', '`' + this.getUrl(linkPrefix + '/detail/${item.id}') + '`'),
        '150px',
      ),
      BaseUI.tableHeader('description', '描述'),
      BaseUI.tableHeaderTime('createdAt', '创建时间'),
    ];
    
    // 获取数据
    const list = await this.get(`${Base.ROOT}/modules`);
    
    return Base.successWrapper(
      BaseUI.tableWithActions(
        this.name,
        '模块',
        await this.buttons(this.name, linkPrefix),
        BaseUI.tableSelect(header, list, 1, [[3, 'desc']]),
        undefined, undefined, '模块',
      ),
      undefined,
      ['text-break'],
    );
  }

  private async buttons(id: string, linkPrefix: string) {
    const prefix = id + '_button_';
    return {
      [`${prefix}refresh`]: BaseUI.refreshButton(),
      [`${prefix}delete`]:  this.currentAuth().admin && BaseUI.deleteButton(id, '删除'),
      [`${prefix}create`]:  BaseUI.linkButton('创建', this.getUrl(`${linkPrefix}/create`)),
    };
  }

  @Post('/list/submit')
  public async submit(@Body() body: any) {
    const ids = this.tableSelected(body[this.name]);
    if (body.op === `${this.name}_button_delete`) {
      for (const id of ids.split(',')) {
        await this.delete(`${Base.ROOT}/modules/${id}`);
      }
    }
    return Base.success(this.getUrl('/module'));
  }

  @Get('/list')
  public async getFormUris() { return Base.successForm(); }
  
  @Get('/list/formId')
  public async formId() { return Base.success(this.name + '_form'); }
  
  @Post('/list/validate')
  public async validate() { return Base.success(null); }
}
```
