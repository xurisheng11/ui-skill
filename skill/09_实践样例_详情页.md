# 实践样例：详情页完整模板

包含页面标题、基本信息卡片、多 Tab 内容的完整详情页。

## TypeScript 实现（Controller）

```typescript
@Controller('/module/detail', { tagName: 'module', description: '模块详情' })
export class ModuleDetailController extends BaseController {
  private readonly name = 'module_detail';

  @Get('/build')
  public async build(@Query('id') id: string, @Query('tab') tab: string) {
    const linkPrefix = `/module/detail/${id}`;
    const data = await this.get(`${Base.ROOT}/modules/${id}`);
    const canWrite = this.currentAuth().admin;
    
    return Base.successWrapper(
      BaseUI.spaceBetween([
        {
          // H1 标题
          basic_title: BaseUI.pageTitle('basic_title', data.name),
          
          // 基本信息卡片（带编辑按钮）
          basic_info: BaseUI.cardPanelWithEdit(
            '基本信息',
            this.getUrl(`${linkPrefix}/edit`),
            {
              name:        BaseUI.cardItem('名称', data.name),
              description: BaseUI.cardItem('描述', data.description),
              type:        BaseUI.cardItem('类型', data.type),
              createdAt:   BaseUI.cardItemTime('创建时间', data.createdAt),
              createdBy:   BaseUI.cardItem('创建人', data.createdBy),
              updatedAt:   BaseUI.cardItemTime('最后修改时间', data.updatedAt),
            },
            canWrite,     // 是否显示编辑按钮
          ),
          
          // Tab 组
          module_tabs: BaseUI.tabs(
            {
              // Tab 1: 数据列表
              list_tab: BaseUI.tabItem('list_tab', '子列表',
                BaseUI.spaceBetween({
                  list_table: BaseUI.tableWithActions(
                    'child_list',
                    '子条目',
                    {
                      child_list_button_refresh: BaseUI.refreshButton(),
                      child_list_button_create: canWrite && BaseUI.linkButton(
                        '添加', this.getUrl(`${linkPrefix}/child/create`)
                      ),
                    },
                    BaseUI.tableSelect(
                      [
                        BaseUI.tableHeader('name', '名称'),
                        BaseUI.tableHeaderTime('createdAt', '创建时间'),
                      ],
                      data.children || [],
                      -1,  // 无选框
                    ),
                    undefined, true, // noCheck=true（不自动加删除弹窗）
                  ),
                }),
              ),
              
              // Tab 2: 配置信息
              config_tab: BaseUI.tabItem('config_tab', '配置',
                BaseUI.spaceBetween({
                  config_info: BaseUI.cardPanelWithEdit(
                    '配置信息',
                    this.getUrl(`${linkPrefix}/config/edit`),
                    {
                      maxConcurrency: BaseUI.cardItem('最大并发', data.config?.maxConcurrency),
                      timeout: BaseUI.cardItem('超时时间(秒)', data.config?.timeout),
                    },
                    canWrite,
                  ),
                }),
              ),
            },
            tab || 'list_tab', // 默认 Tab
          ),
        },
      ]),
      [
        this.loadTabDataJs(id),
      ],
      ['text-break'],
    );
  }

  // 切换 Tab 时加载对应数据
  private loadTabDataJs(id: string) {
    return `
      useEffect(() => {
        if (value_module_tabs === 'list_tab') {
          $.post('/ajax-callback', {method:'get', url:'{LUBAN_HOST}/module/detail/${id}/children'},
            function(items) {
              set_init_items_child_list(items);
            }, 'json'
          );
        }
      }, [value_module_tabs]);
    `;
  }

  @Get('/')
  public async getFormUris() { return Base.successForm(); }
  
  @Get('/formId')
  public async formId() { return Base.success(this.name + '_form'); }
  
  @Post('/validate')
  public async validate() { return Base.success(null); }
}
```
